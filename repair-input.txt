class DashboardScreen extends StatefulWidget {
  const DashboardScreen({super.key});

  @override
  State<DashboardScreen> createState() => _DashboardScreenState();
}

class _DashboardScreenState extends State<DashboardScreen> {
  final ImportService _importService = ImportService();
  final CnnService _cnnService = CnnService();

  List<TelemetrySample> _samples = [];
  List<String> _validationErrors = [];
  List<Map<String, dynamic>> _detectedAnomalies = [];
  bool _isLoading = false;
  bool _modelReady = false;

  @override
  void initState() {
    super.initState();
    _initModel();
  }

  Future<void> _initModel() async {
    await _cnnService.loadModel();
    setState(() {
      _modelReady = _cnnService.isLoaded;
    });
  }

  // Load sample dataset bundled in assets
  Future<void> _loadSampleDataset() async {
    setState(() => _isLoading = true);
    try {
      String rawCsv = await DefaultAssetBundle.of(context).loadString('assets/samples/telemetry_sample.csv');
      var result = await _importService.parseCsv(rawCsv);
      
      setState(() {
        _samples = result.samples;
        _validationErrors = result.errors;
      });

      _runInference();
    } catch (e) {
      setState(() {
        _validationErrors = ['Failed to load sample dataset: $e'];
      });
    } finally {
      setState(() => _isLoading = false);
    }
  }

  void _runInference() {
    if (!_modelReady || _samples.length < 50) return;

    List<Map<String, dynamic>> anomalies = [];
    // Sliding window of 50 samples
    for (int i = 0; i <= _samples.length - 50; i++) {
      List<TelemetrySample> window = _samples.sublist(i, i + 50);
      try {
        double score = _cnnService.predict(window);
        if (score >= 0.5) {
          anomalies.add({
            'index': i + 49,
            'timestamp': window.last.timestamp,
            'score': score,
          });
        }
      } catch (e) {
        // Skip invalid windows
      }
    }

    setState(() {
      _detectedAnomalies = anomalies;
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('GridSentinel - Industrial Telemetry Analysis'),
        actions: [
          Center(
            child: Padding(
              padding: const EdgeInsets.symmetric(horizontal: 16.0),
              child: Chip(
                backgroundColor: _modelReady ? Colors.green[100] : Colors.orange[100],
                label: Text(
                  _modelReady ? '1D-CNN Model Active' : 'Model Loading...',
                  style: TextStyle(color: _modelReady ? Colors.green[800] : Colors.orange[800]),
                ),
              ),
            ),
          )
        ],
      ),
      body: Padding(
        padding: const EdgeInsets.all(16.0),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Row(
              children: [
                ElevatedButton.icon(
                  onPressed: _isLoading ? null : _loadSampleDataset,
                  icon: const Icon(Icons.folder_open),
                  label: const Text('Load Sample Telemetry CSV'),
                ),
                const SizedBox(width: 16),
                Text('Loaded Samples: ${_samples.length}'),
                const SizedBox(width: 16),
                Text('Anomalies Detected: ${_detectedAnomalies.length}', 
                  style: TextStyle(fontWeight: FontWeight.bold, color: _detectedAnomalies.isNotEmpty ? Colors.red : Colors.green)),
              ],
            ),
            const SizedBox(height: 16),
            if (_validationErrors.isNotEmpty)
              Container(
                padding: const EdgeInsets.all(8),
                color: Colors.red[50],
                child: Text('Validation Errors: ${_validationErrors.join(", ")}', style: const TextStyle(color: Colors.red)),
              ),
            const SizedBox(height: 16),
            Expanded(
              flex: 3,
              child: Card(
                elevation: 3,
                child: Padding(
                  padding: const EdgeInsets.all(12.0),
                  child: _samples.isEmpty
                      ? const Center(child: Text('No telemetry loaded. Click the button above to load sensor streams.'))
                      : LineChart(
                          LineChartData(
                            gridData: const FlGridData(show: true),
                            titlesData: const FlTitlesData(show: true),
                            borderData: FlBorderData(show: true),
                            lineBarsData: [
                              LineChartBarData(
                                spots: _samples.asMap().entries.map((e) => FlSpot(e.key.toDouble(), e.value.pressureKpa)).toList(),
                                isCurved: true,
                                color: Colors.blue,
                                barWidth: 2,

class _ImportScreenState extends State<ImportScreen> {
  String _status = 'Select a CSV file or load the sample to validate telemetry data.';
  List<String> _headers = [];
  List<List<dynamic>> _previewData = [];
  bool _isLoading = false;

  void _handleImportResult(Map<String, dynamic> result) {
    setState(() {
      _isLoading = false;
      if (result['success']) {
        _status = 'Validation passed! Loaded ${result['total_rows']} records ready for analysis.';
        _headers = List<String>.from(result['headers']);
        _previewData = List<List<dynamic>>.from(result['data'].take(5));
      } else {
        _status = 'Error: ${result['error']}';
        _headers = [];
        _previewData = [];
      }
    });
  }

  Future<void> _pickFile() async {
    setState(() => _isLoading = true);
    final result = await ImportService.pickAndParseCsv();
    _handleImportResult(result);
  }

  Future<void> _loadSample() async {
    setState(() => _isLoading = true);
    final result = await ImportService.loadSampleCsv();
    _handleImportResult(result);
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(

import 'dart:convert';
import 'dart:io';
import 'package:csv/csv.dart';
import 'package:file_picker/file_picker.dart';
import 'package:flutter/services.dart' show rootBundle;
import '../models.dart';

class ImportResult {
  const ImportResult.error(this.message)
      : ok = false,
        name = '',
        rows = const [];
  const ImportResult.success(this.name, this.rows)
      : ok = true,
        message = '';
  final bool ok;
  final String message;
  final String name;
  final List<Telemetry> rows;
}

class ImportService {
  static const keys = ['pressure_kpa', 'motor_rpm', 'temperature_c', 'flow_lpm'];

  static Future<ImportResult> pickFile() async {
    try {
      final result = await FilePicker.platform.pickFiles(
        type: FileType.custom,
        allowedExtensions: ['csv', 'json'],
      );
      if (result == null || result.files.single.path == null) {
        return const ImportResult.error('No file selected.');
      }
      final name = result.files.single.name;
      final text = await File(result.files.single.path!).readAsString();
      return name.toLowerCase().endsWith('.json')
          ? parseJson(text, name)
          : parseCsv(text, name);
    } catch (e) {
      return ImportResult.error('Could not read file: $e');
    }
  }

  static Future<ImportResult> loadSample() async {
    try {
      final text = await rootBundle.loadString('assets/samples/sample_telemetry.csv');
      return parseCsv(text, 'sample_telemetry.csv');
    } catch (e) {
      return ImportResult.error('Could not load sample: $e');
    }
  }

  static ImportResult parseCsv(String text, String name) {
    final clean = text.replaceAll('\r\n', '\n').replaceAll('\r', '\n');
    final rows = const CsvToListConverter(eol: '\n').convert(clean);
    if (rows.length < 2) return const ImportResult.error('CSV has no data rows.');
    final headers = rows.first.map((e) => e.toString().trim().toLowerCase()).toList();
    final recs = <Map<String, dynamic>>[];
    for (final row in rows.skip(1)) {
      if (row.isEmpty || (row.length == 1 && row.first.toString().trim().isEmpty)) continue;
      final m = <String, dynamic>{};
      for (var i = 0; i < headers.length && i < row.length; i++) {
        m[headers[i]] = row[i];
      }
      recs.add(m);
    }
    return _build(recs, name);
  }

  static ImportResult parseJson(String text, String name) {
    try {
      dynamic data = jsonDecode(text);
      if (data is Map) data = data['data'] ?? data['records'] ?? data['rows'];
      if (data is! List) {
        return const ImportResult.error('JSON must be a list of records (or an object with a data list).');
      }
      final recs = <Map<String, dynamic>>[];
      for (final item in data) {
        if (item is Map) {
          recs.add(item.map((k, v) => MapEntry(k.toString().toLowerCase(), v)));
        }
      }
      return _build(recs, name);
    } catch (e) {
      return ImportResult.error('Invalid JSON: $e');
    }
  }

  static ImportResult _build(List<Map<String, dynamic>> recs, String name) {
    if (recs.isEmpty) return const ImportResult.error('File has no data rows.');
    final missing = keys.where((k) => !recs.first.containsKey(k)).toList();
    if (missing.isNotEmpty) {
      return ImportResult.error('Missing column(s): ${missing.join(', ')}');
    }
    final base = DateTime.now();
    final out = <Telemetry>[];
    var skipped = 0;
    for (var i = 0; i < recs.length; i++) {
      final r = recs[i];
      final vals = keys.map((k) => double.tryParse('${r[k]}')).toList();
      if (vals.any((v) => v == null)) {
        skipped++;
        continue;
      }
      final ts = DateTime.tryParse('${r['timestamp']}') ?? base.add(Duration(seconds: i));
      out.add(Telemetry(ts, vals.map((v) => v!).toList()));
    }
    if (out.length < 30) {
      return ImportResult.error('Need at least 30 valid rows (found ${out.length}, skipped $skipped).');
    }
    return ImportResult.success(skipped > 0 ? '$name ($skipped bad rows skipped)' : name, out);
  }
}
